2. Exact Firebase Setup Instructions:

  1. Create Project: Open Firebase Console, click Add Project and name it the-east-school-jehangira.
  2. Enable Authentication: Go to Build → Authentication → Sign-in method:
    • Enable Email/Password provider.
    • Enable Google provider and configure support email.
  3. Authorized Domains: Under Authentication Settings, add your production domain and localhost.
  4. Firestore Database: Create database in production mode. Set rules: 
    rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /reviews/{reviewId} {
      allow read: if true;
      allow create: if request.auth != null && request.resource.data.comment.size() > 5;
    }
    match /gallery/{photoId} {
      allow read: if true;
      allow create: if request.auth != null;
    }
  }
}
  5. Storage Rules: Set Firebase Storage rules: 
    rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /campus_photos/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null && request.resource.size < 10 * 1024 * 1024;
    }
  }
}
